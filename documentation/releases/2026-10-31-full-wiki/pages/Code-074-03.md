function niceTicks(lo,hi,n){
  if(!Number.isFinite(lo)||!Number.isFinite(hi))return [];
  if(lo===hi){lo-=1;hi+=1;}
  const step0=(hi-lo)/(n||5),mag=Math.pow(10,Math.floor(Math.log10(step0))),err=step0/mag;
